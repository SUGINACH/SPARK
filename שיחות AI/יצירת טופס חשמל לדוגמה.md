    async function encryptJSON(obj) {
        const enc = new TextEncoder();
        const keyMaterial = await crypto.subtle.importKey("raw", enc.encode(ENCRYPTION_KEY), "AES-GCM", false, ["encrypt"]);
        const iv = crypto.getRandomValues(new Uint8Array(12));
        const cipher = await crypto.subtle.encrypt({name: "AES-GCM", iv}, keyMaterial, enc.encode(JSON.stringify(obj)));
        
        const combined = new Uint8Array(12 + cipher.byteLength);
        combined.set(iv, 0);
        combined.set(new Uint8Array(cipher), 12);
        
        let binary = '';
        for(let i=0; i<combined.byteLength; i++) {
            binary += String.fromCharCode(combined[i]);
        }
        return btoa(binary);
    }

    function confirmPeriods() {

                const e = row.querySelector('.g-end').value;
                if(s && e) ranges.push({ start: s.slice(5), end: e.slice(5) });
            });

            periodObj.ranges = ranges;
            btnText = ranges.length > 1 ? "מספר טווחים קלנדריים" : "קלנדרי מותאם";

        else if (type === 'hebrew') {

            document.querySelectorAll('#hebrewRangesList .range-row').forEach(row => {
                const sm = row.querySelector('.h-m-start').value.padStart(2,'0');
                const sd = row.querySelector('.h-d-start').value.padStart(2,'0');
                const em = row.querySelector('.h-m-end').value.padStart(2,'0');
                const ed = row.querySelector('.h-d-end').value.padStart(2,'0');

            });
            if(ranges.length === 0) { alert("נא להזין לפחות טווח אחד תקין."); return; }
            periodObj.ranges = ranges;
            btnText = ranges.length > 1 ? "מספר טווחים עבריים" : "עברי מותאם";
        } 
        else if (type === 'school_year') { btnText = "שנת לימודים"; } 
        else if (type === 'yeshiva') { btnText = "שנת ישיבה"; }

        document.getElementById(currentRowForModal).dataset.periods = JSON.stringify(periodObj);
        updateButtonLabel('periods', currentRowForModal, btnText);
        closeModal('periodsModal');
    }

    async function exportToFile() {
        const clientDetails = {
            name: document.getElementById('client_name').value,
            phone: document.getElementById('client_phone').value,
            address: document.getElementById('client_address').value,
            contract: document.getElementById('client_contract').value,
            filledBy: document.getElementById('client_filledBy').value,
            notes: document.getElementById('client_notes').value
        };

        const locations = [];
        document.querySelectorAll('#locations-container .card').forEach(locDiv => {
            const locName = locDiv.querySelector('.location-title-input').value;
            const locDays = safeParse(locDiv.dataset.days, '[0,1,2,3,4,5,6]');
            const locHours = safeParse(locDiv.dataset.hours, '[]');
            const locPeriods = safeParse(locDiv.dataset.periods, '{"type":"all_year"}');
            
            const consumers = [];
            locDiv.querySelectorAll('.item-row').forEach(row => {
                const typeSelect = row.querySelector('.col-type select');
                const typeTextInput = row.querySelector('.col-type input[type="text"]');
                let category = null, type = null;
                if (typeSelect) {
                    const onchangeAttr = typeSelect.getAttribute('onchange') || '';
                    const m = onchangeAttr.match(/updateDynamicFields\(this, '([^']+)'/);
                    category = m ? m[1] : null;
                    type = typeSelect.value;
                } else if (typeTextInput) {
                    category = 'אחר';
                    type = typeTextInput.value;
                }
                if (!type) return; 

                const fields = {};
                row.querySelectorAll('input[data-field-key]').forEach(inp => {
                    fields[inp.dataset.fieldKey] = parseFloat(inp.value) || 0;
                });

                const catDiv = row.closest('.category-wrapper');
                const catDays = safeParse(catDiv.dataset.days, '[0,1,2,3,4,5,6]');
                const catHours = safeParse(catDiv.dataset.hours, '[]');
                const catPeriods = safeParse(catDiv.dataset.periods, '{"type":"all_year"}');

                const days = safeParse(row.dataset.days, '[0,1,2,3,4,5,6]');
                const hours = safeParse(row.dataset.hours, '[]');
                const periods = safeParse(row.dataset.periods, '{"type":"all_year"}');

                consumers.push({ 
                    category, type, fields, 
                    days, hours, periods,
                    catDays, catHours, catPeriods,
                    locDays, locHours, locPeriods 
                });
            });
            locations.push({ name: locName, consumers });
        });

        const exportObj = {
            formVersion: '3.0',
            exportedAt: new Date().toISOString(),
            clientDetails,
            locations
        };

        try {
            const base64Encrypted = await encryptJSON(exportObj);
            const blob = new Blob([base64Encrypted], { type: 'text/plain' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            const safeName = (clientDetails.name || 'לקוח').replace(/[^\wא-ת]+/g, '_');
            const dateStr = new Date().toISOString().slice(0,10);
            a.href = url;
            a.download = `נתוני_צריכה_${safeName}_${dateStr}.json.enc`;
            document.body.appendChild(a);
            a.click();
            document.body.removeChild(a);
            URL.revokeObjectURL(url);
        } catch (e) {
            alert("שגיאה בהצפנת הקובץ.");
            console.error(e);
        }
    }
