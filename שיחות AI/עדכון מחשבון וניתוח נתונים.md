    function parseIsraelDate(dateStr, timeStr) {

    function calculateMedian(arr) {
        if (arr.length === 0) return 0;
        const sorted = [...arr].sort((a, b) => a - b);
        const mid = Math.floor(sorted.length / 2);
        return sorted.length % 2 !== 0 ? sorted[mid] : (sorted[mid - 1] + sorted[mid]) / 2;
    }

    function calculateTrimmedMean(arr, pct = 0.15) {
        if (arr.length === 0) return 0;
        const sorted = [...arr].sort((a, b) => a - b);
        const cut = Math.round(sorted.length * pct);
        let trimmed = sorted.slice(cut, sorted.length - cut);
        if (trimmed.length === 0) trimmed = sorted;
        return trimmed.reduce((sum, val) => sum + val, 0) / trimmed.length;
    }

    function calculateTopPercentAvg(arr, pct = 0.15) {
        if (arr.length === 0) return 0;
        const sorted = [...arr].sort((a, b) => b - a);
        const count = Math.max(1, Math.round(sorted.length * pct));
        return sorted.slice(0, count).reduce((sum, val) => sum + val, 0) / count;
    }

    function calculateBottomPercentAvg(arr, pct = 0.15) {
        if (arr.length === 0) return 0;
        const sorted = [...arr].sort((a, b) => a - b);
        const count = Math.max(1, Math.round(sorted.length * pct));
        return sorted.slice(0, count).reduce((sum, val) => sum + val, 0) / count;
    }

    function pmt(rate, nper, pv) {
        if (rate === 0) return -(pv / nper);
        return -(rate * pv) / (1 - Math.pow(1 + rate, -nper));
    }

    function npv(rate, cashFlows) {
        return cashFlows.reduce((acc, val, i) => acc + val / Math.pow(1 + rate, i), 0);
    }

    function irr(cashFlows, guess = 0.1) {
        let x0 = guess;
        for (let i = 0; i < 100; i++) {
            const npvVal = npv(x0, cashFlows);
            const npvDeriv = cashFlows.reduce((acc, val, j) => acc - (j * val) / Math.pow(1 + x0, j + 1), 0);
            const x1 = x0 - npvVal / npvDeriv;
            if (Math.abs(x1 - x0) < 1e-7) return x1;
            x0 = x1;
        }
        return x0;
    }

    function runFinancialModel(pAc, pDc, batKwh, annualYield, tariff, inputs) {
        const pvCost = pAc * inputs.capexPV;
        const batCost = batKwh * inputs.capexBattery;
        let capex = pvCost + batCost;
        if (inputs.taxStatus === 'business') capex /= 1.17; 

        const loanAmount = capex * inputs.loanPct;
        const equity = capex - loanAmount;
        const pmtYearly = Math.abs(pmt(inputs.loanInt, inputs.loanYrs, loanAmount));

        let cashFlows = [-equity]; 
        let totalProfit = 0;

        for (let year = 1; year <= 25; year++) {
            let energyProduced = annualYield * Math.pow(1 - CALC_CONSTANTS.degradation, year - 1);
            let revSolar = inputs.zeroExport ? (energyProduced * 0.4 * CALC_CONSTANTS.tou.summer_peak) : (energyProduced * tariff);
            let revArb = 0;
            let revGenSavings = 0;

            if (batKwh > 0) {
                const usableBat = batKwh * CALC_CONSTANTS.battery.dod * CALC_CONSTANTS.battery.eff_discharge;
                revArb = 300 * usableBat * (CALC_CONSTANTS.tou.summer_peak - CALC_CONSTANTS.tou.summer_low);
                
                if (inputs.ignoreMissingShabbat && (inputs.useStorageShabbat || inputs.useStorageHoliday)) {
                    revGenSavings = (50 * inputs.estShabbatLoad * inputs.genCost);
                }
            }

            let revenue = revSolar + revArb + revGenSavings;
            let opex = (pAc * CALC_CONSTANTS.opex.pv) + (batKwh * CALC_CONSTANTS.opex.battery);
            let financing = (year <= inputs.loanYrs) ? pmtYearly : 0;
            
            let replacement = 0;
            if (year === CALC_CONSTANTS.replacements.inverter_year) replacement += pvCost * CALC_CONSTANTS.replacements.inverter_cost_pct;
            if (batKwh > 0 && year === CALC_CONSTANTS.replacements.battery_year) replacement += batCost * CALC_CONSTANTS.replacements.battery_cost_pct;

            let tax = 0;
            let taxableIncome = revenue - opex;
            
            if (inputs.taxStatus === 'private') {
                if (taxableIncome > 25000) tax = (taxableIncome - 25000) * 0.10;
            } else {
                let depreciation = (year <= 4) ? (capex * 0.25) : 0;
                taxableIncome = revenue - opex - depreciation - financing;
                if (taxableIncome > 0) tax = taxableIncome * 0.23;
            }

            let netCashFlow = revenue - opex - replacement - tax - financing;
            cashFlows.push(netCashFlow);
            totalProfit += netCashFlow;
        }

        return {
            capex: capex,
            equity: equity,
            capacity: pAc,
            batKwh: batKwh,
            totalProfit: totalProfit,
            irr: (irr(cashFlows) * 100).toFixed(2),
            npv: npv(0.05, cashFlows).toFixed(0)
        };
    }

    <h2>2. רשימת כל היכולות והפיצ'רים שביקשתי</h2>
    <ul>
        <li><strong>קליטה ועיבוד קובץ CSV:</strong> העלאת קובץ צריכה של חברת החשמל, חילוץ אוטומטי של נתוני הלקוח (שם, כתובת, מס' חוזה, פרטי מונה) ועיבוד של כל שורות הצריכה לפי תאריך ושעה.</li>
        <li><strong>זיהוי תעו"ז (פסגה/שפל):</strong> המערכת מזהה אילו שעות נחשבות "פסגה" לפי חוקי רשות החשמל, כולל התעלמות מפסגות בסופי שבוע (שישי-שבת), איפוס פסגות בעונות מעבר, וזיהוי מדויק של חגי ישראל (קודד מערך של תאריכי חגים קבועים מראש בין השנים 2020-2030 כדי שלא ייחשבו כימי פסגה בקיץ).</li>
        <li><strong>הפקת דוחות (ממשק + אקסל):</strong> המערכת מייצרת 3 טבלאות נתונים הניתנות להורדה לאקסל: דו"ח שעתי, דו"ח יומי מרוכז, ודו"ח ימי פסגה בלבד כולל תמחור.</li>
        <li><strong>ניתוח סטטיסטי של ימי הפסגה:</strong> המערכת מחשבת חציונים, ממוצעים חתוכים (קיצוץ 15% עליונים ותחתונים), וממוצעים של קצוות (15% עליונים/תחתונים). הניתוח מפוצל ל-3 קטגוריות: כללי (קיץ+חורף), חורף בלבד, וקיץ בלבד.</li>
        <li><strong>חישוב הנדסי סולארי:</strong> חישוב גודל מערכת נדרשת לפי שטח גג, המרת DC ל-AC, שקלול כיוון גג (אזימוט) והצללות, וקביעת תעריף המכירה לרשת (אסדרת יצרן מול צרכן).</li>
        <li><strong>מודל אגירה ו"חשמל כשר":</strong> תכנון גודל סוללה אופטימלי, עם אפשרות לדריסות נפח בהתאם לדרישות גיבוי לשבתות וחגים, וחישוב חיסכון של סולר (גנרטור) במקרה של ניתוק יזום מהרשת בשבת.</li>
        <li><strong>סימולציה פיננסית (25 שנה):</strong> בניית תזרים מזומנים מלא הכולל: עלויות הקמה (CAPEX), הלוואה בנקאית (שפיצר), ירידת תפוקה שנתית (Degradation), רווחי ארביטראז' מהסוללה, הוצאות תפעול, החלפות ציוד (ממיר וסוללה בשנים 14/15), והשפעות מסוי (פרטי מול חברה). הצגת ROI, NPV ו-IRR.</li>
        <li><strong>הגדרות תעריפים דינמיות:</strong> מודל (חלון קופץ) המאפשר למשתמש לשנות את תעריפי הפסגה והשפל לכל עונה, לפני הרצת הסימולציה.</li>
    </ul>

    <h2>4. כל החישובים והנוסחאות</h2>
    <h3>א. זיהוי שעות פסגה (לוגיקת isPeakHour)</h3>
    <p>
        <strong>חורף (חודשים 12, 1, 2):</strong> פסגה בין השעות 17:00 עד 21:59 (ללא החרגת סופ"ש).<br>
        <strong>מעבר (חודשים 3, 4, 5, 10, 11):</strong> אין שעות פסגה כלל.<br>
        <strong>קיץ (חודשים 6, 7, 8, 9):</strong> פסגה בין 17:00 ל-22:59, <strong>אך מוחרגים:</strong> ימי שישי (wday=5), ימי שבת (wday=6), ורשימה קשיחה של תאריכים (YYYY-MM-DD) המייצגים חגי ישראל בין 2020 ל-2030 (ראש השנה, סוכות, פסח, שבועות וכו').
    </p>

    <h3>ב. סטטיסטיקות (על מערך ימי הפסגה)</h3>
    <p>החישובים מבוצעים לאחר מיון המערך (מהקטן לגדול או להפך):</p>
    <ul>
        <li><strong>חציון:</strong> אם אורך המערך אי-זוגי, נלקח הערך האמצעי. אם זוגי, ממוצע של שני הערכים האמצעיים.</li>
        <li><strong>ממוצע חתוך 15% (Trimmed Mean):</strong> חיתוך של 15% מגודל המערך משני הקצוות (Floor של `length * 0.15`), וחישוב ממוצע של שאר האיברים. אם לאחר החיתוך אין איברים, יוחזר הממוצע הרגיל.</li>
        <li><strong>ממוצע 15% עליונים/תחתונים:</strong> לקיחת `count` איברים (Max של 1 או `Round(length * 0.15)`) מהקצה הרלוונטי וחישוב ממוצע שלהם.</li>
    </ul>

    <h3>ג. הנדסה ומידות (Solar Sizing)</h3>
    <p>
        <code>pDc = RoofArea / 5.5</code> (יחס שטח להספק)<br>
        <code>pAc = pDc / 1.3</code> (יחס העמסה AC/DC)<br>
        <code>AnnualYield = pDc * 1750 * Azimuth * Shading</code>
    </p>

    <h3>ד. תעריפי מכירה (FIT)</h3>
    <p>מחושב כממוצע משוקלל בהתאם לגודל המערכת (כולל מערכת קיימת P0):</p>
    <ul>
        <li>עד 15kW: 0.48 ₪</li>
        <li>מ-15kW עד 100kW: 0.3731 ₪</li>
        <li>מעל 100kW: 0.3437 ₪</li>
        <li>תוספת פרמיה אורבנית (אם סומן): +0.06 ₪.</li>
        <li>אם סומן <strong>Zero Export:</strong> התעריף מתאפס, וההנחה היא צריכה עצמית המוערכת לפי `AnnualYield * 0.4 * Summer_Peak_Rate`.</li>
    </ul>

    <h3>ה. גודל סוללה וארביטראז'</h3>
    <p>
        גודל סוללה בסיסי: <code>BatteryKwh = pAc * 1.5</code>.<br>
        אם נבחר גיבוי שבת: סוללה תהיה המקסימום בין הבסיס לבין `EstShabbatLoad * 1.2`.<br>
        אם נבחר גיבוי חג: סוללה תהיה המקסימום בין הבסיס לבין `EstShabbatLoad * 2.2`.<br>
        <strong>רווח ארביטראז' שנתי (טעינה בשפל ופריקה בפסגה):</strong> `300 ימים * (BatteryKwh * 0.95 DoD * 0.94 Dis_Eff) * (Summer_Peak - Summer_Low)`.
    </p>

    <h3>ו. מימון ותזרים מזומנים (Cash Flow)</h3>
    <ul>
        <li>הלוואה: חישוב תשלום שנתי קבוע באמצעות נוסחת `PMT`.</li>
        <li>מס הכנסה אדם פרטי: 10% מס רק על הכנסה שמעל 25,000 ₪ (Revenue - Opex).</li>
        <li>מס חברה: הפחתת מימון ופחת מואץ (25% לשנה ב-4 שנים הראשונות), ואז מס חברות של 23% על הרווח. ה-CAPEX מחולק ב-1.17 לקיזוז מע"מ.</li>
        <li>החלפות: ממיר עולה 15% מה-CAPEX הסולארי בשנה 14. סוללה עולה 60% מ-CAPEX הסוללה בשנה 15.</li>
        <li>מדדי כדאיות מחושבים בסוף: NPV לפי 5% הוון (`npv(0.05, cashFlows)`) ו-IRR (`irr(cashFlows) * 100`).</li>
    </ul>

    <h2>5. כל התיקונים ושינויי הכיוון שביקשתי במהלך הדרך</h2>

    <h2>6. מבנה הנתונים</h2>
    <p>האובייקטים המרכזיים שמנהלים את הסטייט במערכת (Client Side):</p>
    <ul>
        <li><code>hourlyAggregation</code>: אובייקט מילוני. מפתח: מחרוזת תאריך+שעה (DD/MM/YYYY HH:00). ערך: `{ kwh: Number, day: String }`.</li>
        <li><code>dailyAggregation</code>: אובייקט מילוני. מפתח: מחרוזת תאריך (DD/MM/YYYY). ערך: `{ kwh: Number, day: String }`.</li>
        <li><code>dailyTouAggregation</code>: אובייקט מילוני. מפתח: מחרוזת תאריך. ערך: `{ peak: Number, offPeak: Number, day: String }`.</li>
        <li><code>globalCalculatedStats</code>: אובייקט המכיל 3 מפתחות: <code>combined, winter, summer</code>. כל אחד מהם מכיל אובייקט עם 6 שדות מסוג Number (medianKwh, medianCost, trimmedKwh, trimmedCost, top15Kwh, bottom15Kwh).</li>
        <li><code>globalSimulationData</code>: נשמר כרקורד JSON ל-LocalStorage. מכיל: `{ pvOnly: Object, withBat: Object, inputs: Object }`. כל Object מכיל פירוט מלא של התוצאה הפיננסית.</li>
    </ul>
