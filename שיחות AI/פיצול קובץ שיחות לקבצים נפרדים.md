### 🤖 מודל

הנה קוד פייתון שמיועד לבצע את המשימה הזו. 

הקוד קורא את הקובץ המרכזי, מחלק אותו לפי שורות ההפרדה (`===`), מחלץ מכל חלק את הכותרת הראשונה (שמתחילה ב-`#`) כדי להשתמש בה כשם הקובץ, מנקה תווים שאינם חוקיים לשמירה במערכת ההפעלה, ושומר כל שיחה כקובץ Markdown נפרד.

### קוד הפייתון:

```python
import os
import re

def sanitize_filename(filename, replacement="_"):
    """
    מנקה את כותרת השיחה מתווים שאינם חוקיים לשמירה כשם קובץ במערכת ההפעלה.
    תומך בשמות קבצים בעברית ובאנגלית.
    """
    # תווים אסורים במערכות הפעלה שונות
    invalid_chars = r'[\\/*?:"<>|]'
    sanitized = re.sub(invalid_chars, replacement, filename)
    # הסרת רווחים או נקודות מיותרות בקצוות
    sanitized = sanitized.strip().strip('.')
    # הגבלת אורך שם הקובץ למניעת שגיאות נתיב ארוך
    sanitized = sanitized[:100].strip()
    return sanitized if sanitized else "untitled_conversation"

def split_chat_file(file_path="readable_chat.md"):
    """
    מקבל נתיב לקובץ הצי׳ט המאוחד, מחלק אותו לפי נושאים ושומר כל נושא בקובץ נפרד.
    """
    if not os.path.exists(file_path):
        print(f"שגיאה: הקובץ '{file_path}' לא נמצא בנתיב המבוקש.")
        return

    with open(file_path, 'r', encoding='utf-8') as f:
        content = f.read()

    # ביטוי רגולרי לזיהוי שורת ההפרדה (לפחות 10 סימני שווה רצופים בשורה נפרדת)
    separator_pattern = r'\n?^[ \t]*={10,}[ \t]*$\n?'
    conversations = re.split(separator_pattern, content, flags=re.MULTILINE)

    created_files = []
    title_counters = {}

    for idx, conv in enumerate(conversations):
        conv = conv.strip()
        if not conv:
            continue

        # חיפוש כותרת השיחה הראשונה (שורה שמתחילה ב-#)
        title_match = re.search(r'^#\s+(.+)$', conv, re.MULTILINE)
        if title_match:
            raw_title = title_match.group(1).strip()
        else:
            raw_title = f"שיחה ללא שם {idx + 1}"

        # ניקוי השם ליצירת שם קובץ תקין
        base_title = sanitize_filename(raw_title)
        
        # טיפול במקרים של כותרות כפולות (הוספת מספר סידורי)
        if base_title in title_counters:
            title_counters[base_title] += 1
            filename = f"{base_title}_{title_counters[base_title]}.md"
        else:
            title_counters[base_title] = 1
            filename = f"{base_title}.md"

        # שמירת הקובץ החדש
        try:
            with open(filename, 'w', encoding='utf-8') as out_f:
                out_f.write(conv)
            created_files.append(filename)
        except Exception as e:
            print(f"שגיאה בשמירת הקובץ '{filename}': {e}")

    # הדפסת סיכום הפעולה
    print(f"\nהתהליך הושלם בהצלחה. נוצרו {len(created_files)} קבצים נפרדים:")
    for name in created_files:
        print(f"- {name}")

if __name__ == "__main__":
    # במידה ושם הקובץ שלך שונה, ניתן לשנות את הפרמטר כאן
    split_chat_file("readable_chat.md")
```
