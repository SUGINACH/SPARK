### 👤 משתמש

תוכל לייצר לי דף HTML למילוי נתוני צריכה המחולק לסקשנים נפרדים. 1. מיקום. עם לחצן להוספת מיקום חדש, ועריכת שם המיקום (ברירת מחדל: מיקום 1, מיקום 2 וכו')  2. בכל מיקום יהיו כמה סקשנים לסוגי צרכנים, ואלו הן: מיזוג, תאורה, מטבח, תפעול, משרד, כללי, אחר. תחת כל סקשן סוג צרכן יהיה לחצן + להוספת צרכן, - בלחיצה על הוספת צרכן תופיע שורה - בתחילתה של השורה תהיה רשימת דרופדאון עם הצרכנים מהסוג של אותו סקשן. - בבחירה באחד מהצרכנים יופיעו האפשרויות המתאימות לכל צרכן, כפי שמופיע בטבלה הבאה:

        const icon = document.getElementById(`icon-${catId}`);
        
        if (content.classList.contains('active')) {

            icon.classList.remove('active-icon');
        } else {
            // סגירת כל הקטגוריות *בתוך אותו מיקום*
            document.querySelectorAll(`#content-${locId} .category-content`).forEach(c => c.classList.remove('active'));
            document.querySelectorAll(`#content-${locId} .category-header .toggle-icon`).forEach(i => i.classList.remove('active-icon'));
            
            content.classList.add('active');
            icon.classList.add('active-icon');
        }
    }

    // הוספת צרכן
    function addConsumer(containerId, category) {
        const container = document.getElementById(containerId);
        const rowId = 'row-' + Date.now();
        const row = document.createElement("div");
        row.className = "item-row";
        row.id = rowId;

        let typeHtml = category === 'אחר' ?

        const fields = (selectElem.value && consumerData[category][selectElem.value]) ? consumerData[category][selectElem.value] : [];
        renderDynamicFieldsDirect(rowId, fields);
    }
